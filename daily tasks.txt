	/// Post-UAT dual-date-range comparison layer.
	/// Every measure re-evaluates an existing KPI measure over an explicit date
	/// window resolved by 'Period Control', so the Current Period and the
	/// Comparison Period are fully independent and the Comparison Period may be
	/// any historical range (e.g. Aug 1-7 vs Apr 22-28).
	/// Ratio KPIs (CPA, CVR, Cp*) are recalculated inside each window by the
	/// underlying measure itself - daily ratios are never averaged.
	/// Variance and % Change are gated on [Periods Have Equal Length].
	table 'KPI Comparison'
		isHidden
		lineageTag: ad3c7763-4255-4d2d-961d-9ed4ff375d09

		measure 'Total Cost Current Period' =
				VAR S = [Current Period Start]
				VAR E = [Current Period End]
				RETURN
				    CALCULATE (
				        [Total Cost],
				        REMOVEFILTERS ( 'dim_date' ),
				        'dim_date'[date] >= S,
				        'dim_date'[date] <= E
				    )
			formatString: $#,##0.0
			displayFolder: 20 Comparison\Total Cost
			lineageTag: ad2eb244-c3f1-4210-8ec7-25351e769144

		measure 'Total Cost Comparison Period' =
				VAR S = [Comparison Period Start]
				VAR E = [Comparison Period End]
				RETURN
				    CALCULATE (
				        [Total Cost],
				        REMOVEFILTERS ( 'dim_date' ),
				        'dim_date'[date] >= S,
				        'dim_date'[date] <= E
				    )
			formatString: $#,##0.0
			displayFolder: 20 Comparison\Total Cost
			lineageTag: efb49f6c-2625-4db2-aecc-da891564dd0e

		measure 'Total Cost Variance vs Comparison' =
				IF (
				    [Periods Have Equal Length] = 1,
				    [Total Cost Current Period] - [Total Cost Comparison Period]
				)
			formatString: $#,##0.0
			displayFolder: 20 Comparison\Total Cost
			lineageTag: 6c6dfd24-01df-4871-a1f7-a06c942073f4

		measure 'Total Cost % Change vs Comparison' =
				VAR Cur = [Total Cost Current Period]
				VAR Cmp = [Total Cost Comparison Period]
				RETURN
				    IF (
				        [Periods Have Equal Length] = 1
				            && NOT ISBLANK ( Cmp )
				            && Cmp <> 0,
				        DIVIDE ( Cur - Cmp, ABS ( Cmp ) )
				    )
			formatString: +0.0%;-0.0%;0.0%
			displayFolder: 20 Comparison\Total Cost
			lineageTag: 77757993-f06e-4b0d-8f53-05956f185d48

		measure 'Total Cost Current Label' =
				VAR V = [Total Cost Current Period]
				VAR A = ABS ( V )
				VAR Txt =
				    SWITCH (
				        TRUE (),
				        A >= 1000000, FORMAT ( V / 1000000, "$#,##0.0" ) & "M",
				        A >= 1000, FORMAT ( V / 1000, "$#,##0.0" ) & "K",
				        FORMAT ( V, "$#,##0.00" )
				    )
				RETURN
				    IF ( ISBLANK ( V ), "Current: n/a", "Current: " & Txt )
			displayFolder: 20 Comparison\Total Cost
			lineageTag: 0f9c0a98-f55c-4ddb-9c76-b09dd0f0ba13

		measure 'Total Cost Comparison Label' =
				VAR V = [Total Cost Comparison Period]

				    )
				RETURN
				    IF ( ISBLANK ( V ), "Comparison: n/a", "Comparison: " & Txt )
